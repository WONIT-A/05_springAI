# 이 파일은 들여쓰기(스페이스 2칸)가 곧 문법입니다. 탭을 쓰거나 칸이 어긋나면 앱이 뜨지 않습니다.
# ${이름:기본값} = 환경변수가 있으면 그 값, 없으면 콜론 뒤의 기본값.
# 원래는 비밀번호를 이 파일에 직접 쓰지 않습니다. 깃에 올라가는 순간 모두에게 공개되기 때문입니다.
spring:
  application:
    name: wonit
  datasource:
    #   h2 database
    url: jdbc:h2:mem:testdb
    driver-class-name: org.h2.Driver
    username: sa
    password:

#  # MySQL Database
#      url: jdbc:mysql://${host}:3306/${database}?useSSL=false&serverTimezone=Asia/Seoul
#      username: ${username}
#      password: ${password}
#      driver-class-name: com.mysql.cj.jdbc.Driver
  jpa:
    hibernate:
      # ddl-auto 옵션: create, create-drop, update, validate, none
      # - create: 테이블을 새로 만듭니다. 기존 데이터가 삭제될 수 있습니다.
      # - create-drop: 테이블을 만들고 애플리케이션 종료 시 제거합니다.
      # - update: 기존 테이블 구조를 엔티티에 맞춰 변경하려고 시도합니다.
      # - validate: 구조가 맞는지만 검사합니다.
      # - none: 스키마 관련 자동 처리를 하지 않습니다.
      ddl-auto: create
      database-platform: org.hibernate.dialect.H2Dialect

    show-sql: true            # 실행되는 SQL 을 콘솔에 찍는다
    properties:
      hibernate:
        format_sql: true      # SQL 을 줄바꿈해서 사람이 읽기 좋게 출력한다

    # Hibernate가 테이블을 생성한 후 data.sql 실행
    defer-datasource-initialization: true
  h2:
    console:
      enabled: true

   public AccountResponse createAccount(@RequestBody AccountCreateRequest request) {

   **🔍 디버깅 루틴 7단계**

   1. **증상을 한 문장으로** — "1002-345-112233 을 조회하면 404 가 난다"
   2. **재현** —  어느 요청으로 매번 똑같이 나는가?
   3. **스택트레이스 읽기** — 맨 위는 "어디서 터졌나", **맨 아래 `Caused by:`** 가 "진짜 원인". `com.example.wonit` 으로 시작하는 첫 줄 = 우리 코드
   4. **로그 따라가기** — 마지막으로 찍힌 우리 로그는? 그다음 찍혀야 할 로그는?
   5. **SQL · 바인딩 로그** — DB 에 내가 생각한 값이 갔나?
   6. **브레이크포인트** — 위에서 좁힌 곳에 걸고 변수값 확인
   7. **가설 → 한 군데만 고치기 → 2번으로**
